**Authentication Flow:**

1. User initiates login (email/password or OAuth)
2. Firebase Authentication validates credentials
3. ID token generated by Firebase
4. Token sent to backend for verification
5. Backend verifies token with Firebase Admin SDK
6. JWT issued with user role claims
7. Frontend stores JWT in localStorage
8. Subsequent requests use JWT in Authorization header

// server/middleware/auth.middleware.ts
export function requireRoles(allowedRoles: UserRole[]) {
  return (req, res, next) => {
    const user = req.user;

    if (!allowedRoles.includes(user.role)) {
      return res.status(403).json({ error: 'Insufficient permissions' });
    }

    next();
  };
}

### 2.5 Security Rules (Recommended)

**File:** `firestore.rules` (to be created)

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    function hasRole(role) {
      return isAuthenticated() &&
             request.auth.token.role == role;
    }

    function isAdmin() {
      return hasRole('ADMIN');
    }

    // Crowd density (read: all authenticated, write: backend only)
    match /crowdDensity/{gridId} {
      allow read: if isAuthenticated();
      allow write: if false; // Backend only
    }

    // Incidents
    match /incidents/{incidentId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if hasRole('SECURITY') || hasRole('ADMIN');
      allow delete: if isAdmin();
    }

    // Responders
    match /responders/{responderId} {
      allow read: if isAuthenticated();
      allow write: if request.auth.uid == responderId || isAdmin();
    }

    // Alerts (read: all, write: admins only)
    match /alerts/{alertId} {
      allow read: if isAuthenticated();
      allow write: if isAdmin() || hasRole('SECURITY');
    }

    // Team locations
    match /team_locations/{locationId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if request.auth.uid == resource.data.userId;
    }
  }
}
```

**Subscription on Login:**

```typescript
router.post('/fcm-token', authenticate, async (req, res) => {
  const { fcmToken, platform } = req.body;
  const user = req.user;

  // Update database
  await prisma.user.update({
    where: { id: user.id },
    data: { fcmToken, updatedAt: new Date() },
  });

  // Subscribe to topics
  const topics = [
    `event_${user.eventId}`,
    `event_${user.eventId}_${user.role.toLowerCase()}`,
    `role_${user.role.toLowerCase()}`,
    `user_${user.id}`,
  ];

  for (const topic of topics) {
    await admin.messaging().subscribeToTopic([fcmToken], topic);
  }

  res.json({ success: true });
});
```
